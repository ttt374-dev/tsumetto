// src/features/kif/hooks/useHands.ts
import { useMemo } from "react";
import type { PieceTypeKey, KifData } from "../types";

type HandMap = Partial<Record<PieceTypeKey, number>>;

export function useHands(kif: KifData | null) {
  return useMemo(() => {
    if (!kif) {
      return {
        black: "",
        white: "",
        raw: { black: {}, white: {} }
      };
    }

    const raw = parseHandsFromKif(kif);

    return {
      raw,
      black: formatHands(raw.black),
      white: formatHands(raw.white),
    };
  }, [kif]);
}

const PIECE_LABEL: Record<PieceTypeKey, string> = {
  FU: "歩",
  KYO: "香",
  KEI: "桂",
  GIN: "銀",
  KIN: "金",
  KAKU: "角",
  HI: "飛",
};

const NUM_LABEL = ["", "一", "二", "三", "四", "五", "六", "七", "八", "九"];

function formatHands(hand: HandMap): string {
  return Object.entries(hand)
    .filter(([, n]) => n && n > 0)
    .map(([piece, n]) =>
      n === 1
        ? PIECE_LABEL[piece as PieceType]
        : `${PIECE_LABEL[piece as PieceType]}${NUM_LABEL[n]}`
    )
    .join(" ");
}
